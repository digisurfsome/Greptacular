  <ui_ux_standards>
    <!--
      These are non-negotiable standards extracted from professional production
      development. They apply to ALL features regardless of tech stack.
      Agents MUST follow these - they are not suggestions.
    -->

    <banned_patterns>
      <!-- These fail the build. Never use them. -->
      - alert() - Use toast notifications instead
      - confirm() - Use a ConfirmModal component instead
      - prompt() - Use a proper form/modal instead
      - console.log for user feedback - Use toast/notification system
      - Text-only empty states ("No items") - Use EmptyState component with icon + description + CTA button
      - Browser default dialogs of any kind
      - Loading states that just say "Loading..." - Use skeleton loaders for lists, spinner inside buttons
      - Inline styles - Use the project's styling system (Tailwind, CSS modules, etc.)
      - Delete without confirmation - Always use ConfirmModal for destructive actions
      - Raw timestamps in UI - Always format as relative time ("5m ago", "Yesterday", "Jan 15")
    </banned_patterns>

    <required_components>
      <!-- Every app MUST implement these. They are not optional. -->
      - Modal: base modal with overlay, close button, title, content slots, Escape key to close
      - ConfirmModal: "Are you sure?" dialog for all destructive actions
      - Toast/notification system: slide-in notification (success, error, info variants)
      - Skeleton loader: animated placeholder matching content shape for loading states
      - EmptyState: icon + message + CTA button for empty lists/pages
      - ErrorBoundary: catches crashes, shows fallback UI instead of white screen
      - Button with loading state: shows spinner inside, disables during async actions
      - Avatar with fallback: shows initials when image fails to load
    </required_components>

    <page_patterns>
      <!-- User data MUST follow this navigation pattern -->
      <list_view>Shows all items as cards/rows. Clicking navigates to Detail View. Has "Create New" button.</list_view>
      <detail_view>Read-only display of single item. Action buttons: Edit, Delete, Share. Delete opens ConfirmModal.</detail_view>
      <create_view>Form to create new item. Save navigates to Detail View. Cancel returns to List.</create_view>
      <edit_view>Form pre-filled with existing data. Save returns to Detail View. Cancel returns to Detail View (not List).</edit_view>
      <anti_patterns>
        - NEVER: clicking saved item opens edit mode directly
        - NEVER: single "smart" component handles both view and edit
        - NEVER: create form doubles as edit form by pre-loading data
        - NEVER: no way to view an item without editing it
      </anti_patterns>
    </page_patterns>

    <feedback_protocol>
      <!-- Every user action MUST have appropriate feedback -->
      <on_success>Show success toast + navigate to appropriate view</on_success>
      <on_error>Show error toast + stay on current view + keep form data intact</on_error>
      <on_delete>ConfirmModal -> loading state on button -> success toast + redirect to list OR error toast + close modal</on_delete>
      <on_loading>
        - Lists: skeleton cards matching content shape (NOT spinner)
        - Detail pages: skeleton matching content layout
        - Buttons during action: spinner inside button + disable button
        - Page transitions: brief skeleton or fade
      </on_loading>
    </feedback_protocol>

    <polish_requirements>
      <!-- These details separate amateur apps from professional ones -->
      - Relative date formatting: "Just now", "5m ago", "2h ago", "Yesterday", "3d ago", "Jan 15"
      - Text truncation: long text MUST be truncated (single-line ellipsis or multi-line clamp)
      - Back navigation: every detail/edit page has a back button or breadcrumb
      - Hover states: every clickable element visually responds to hover
      - Focus states: visible focus ring on all interactive elements (for keyboard navigation)
      - Autofocus: first input focused when page/modal loads
      - Pluralization: "1 item" not "1 items" - handle singular/plural correctly
      - Search/filter: any list that can grow beyond 5 items needs search or filter
      - Retry on error: error states show a "Try Again" button, not just an error message
      - Unsaved changes: warn before navigating away from dirty forms
      - 404 handling: both route-level (unknown URLs) and data-level (deleted/missing items)
      - Touch targets: minimum 44x44px for all clickable elements on mobile
      - Dynamic page titles: document.title updates on each page/route
      - Zero console errors: no console.log, no React key warnings, no TypeScript errors
    </polish_requirements>

    <accessibility_basics>
      - Focus rings on all interactive elements (visible, not removed)
      - Escape key closes all modals and overlays
      - aria-label on all icon-only buttons
      - Screen reader text (sr-only) for loading states and status updates
      - Form inputs have associated labels (not just placeholder text)
      - Color is not the only indicator of state (add icons or text alongside color)
      - Minimum color contrast ratios (4.5:1 for text, 3:1 for large text)
    </accessibility_basics>

    <responsive_design>
      <!-- Build mobile-first, then scale up -->
      <breakpoints>
        - Mobile: default styles (no prefix) - less than 640px
        - Tablet: sm:640px and up
        - Desktop: lg:1024px and up
      </breakpoints>
      <layout_rules>
        - Mobile: full-width content, hamburger menu, cards stack vertically
        - Desktop: sidebar navigation (240px), multi-column grids, contained forms
        - Sidebar: hidden on mobile, slide-in overlay with hamburger toggle
        - Modals: nearly full-screen on mobile, centered max-w-md on desktop
        - Buttons: full-width for primary actions on mobile, auto-width on desktop
        - Text: minimum 16px on mobile to prevent zoom on iOS
      </layout_rules>
    </responsive_design>
  </ui_ux_standards>

  <coding_standards>
    <!--
      Code quality rules that prevent common AI-generated code problems.
      These apply regardless of framework or language.

    <architecture>
      - One component per file. Group related components in feature folders.
      - Service layer pattern: all database/API operations go through service files, never called directly from UI components.
      - Separate concerns: UI components handle rendering, hooks handle logic, services handle data.
      - Define TypeScript interfaces for ALL data types - no 'any' types.
      - All database writes include createdAt/updatedAt timestamps.
      - All user data scoped by user ID - no cross-user data leaks.
    </architecture>

    <routing>
      - Public routes (landing, login) are unprotected
      - Authenticated routes wrapped in ProtectedRoute guard
      - Admin routes wrapped in AdminRoute guard (if roles exist)
      - Route-level 404 catch-all for unknown paths
      - Auth redirects: unauthenticated -> login, unauthorized -> dashboard
    </routing>

    <error_handling>
      - ErrorBoundary wraps the entire app to prevent white screen of death
      - Network errors show user-friendly messages, not raw error objects
      - Offline detection with banner when connection lost
      - Session expiry handled gracefully (redirect to login with message)
      - Form submission errors keep form data intact
      - Every error state has a retry action
    </error_handling>
  </coding_standards>
